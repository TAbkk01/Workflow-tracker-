  <script type="module">

    import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";

    import { 

      getAuth, 

      signInAnonymously, 

      signInWithCustomToken, 

      onAuthStateChanged 

    } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";

    import { 

      getFirestore, 

      collection, 

      doc, 

      setDoc, 

      addDoc, 

      updateDoc, 

      deleteDoc, 

      onSnapshot 

    } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

    import { 

      getStorage, 

      ref, 

      uploadBytes, 

      getDownloadURL 

    } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-storage.js";


    // Firebase configuration fallback

    const firebaseConfig = typeof __firebase_config !== 'undefined' 

      ? JSON.parse(__firebase_config) 

      : {

          apiKey: "demo-api-key",

          authDomain: "demo-project.firebaseapp.com",

          projectId: "demo-project",

          storageBucket: "demo-project.appspot.com",

          messagingSenderId: "123456789",

          appId: "1:123456789:web:demo"

        };


    const app = initializeApp(firebaseConfig);

    const auth = getAuth(app);

    const db = getFirestore(app);

    const storage = getStorage(app);

    const appId = typeof __app_id !== 'undefined' ? __app_id : 'shared-workflow-app';


    window.FirebaseServices = {

      auth,

      db,

      storage,

      appId,

      signInAnonymously,

      signInWithCustomToken,

      onAuthStateChanged,

      collection,

      doc,

      setDoc,

      addDoc,

      updateDoc,

      deleteDoc,

      onSnapshot,

      ref,

      uploadBytes,

      getDownloadURL

    };

  </script>


  <script type="text/babel">

    const { useState, useEffect, useMemo, useRef } = React;


    const STAGES = [

      { id: 'backlog', name: 'Backlog', badge: 'bg-slate-200 text-slate-800' },

      { id: 'in_progress', name: 'In Progress', badge: 'bg-blue-100 text-blue-800' },

      { id: 'in_review', name: 'In Review', badge: 'bg-amber-100 text-amber-800' },

      { id: 'completed', name: 'Approved & Completed', badge: 'bg-emerald-100 text-emerald-800' }

    ];


    const PRIORITIES = {

      low: { label: 'Low', color: 'bg-slate-100 text-slate-700 border-slate-200', order: 1 },

      medium: { label: 'Medium', color: 'bg-blue-100 text-blue-700 border-blue-200', order: 2 },

      high: { label: 'High', color: 'bg-amber-100 text-amber-700 border-amber-200', order: 3 },

      urgent: { label: 'Urgent', color: 'bg-rose-100 text-rose-700 border-rose-200', order: 4 }

    };


    const FREQUENCIES = {

      daily: { label: 'Daily', badge: 'bg-indigo-100 text-indigo-700 border-indigo-200' },

      weekly: { label: 'Weekly', badge: 'bg-purple-100 text-purple-700 border-purple-200' },

      monthly: { label: 'Monthly', badge: 'bg-teal-100 text-teal-700 border-teal-200' }

    };


    function App() {

      const [user, setUser] = useState(null);

      const [boardId, setBoardId] = useState('main-team-board');

      const [boardInput, setBoardInput] = useState('');

      const [tasks, setTasks] = useState([]);

      const [loading, setLoading] = useState(true);


      const [activeTab, setActiveTab] = useState('main'); 

      const [viewMode, setViewMode] = useState('kanban'); 

      const [recurringFrequency, setRecurringFrequency] = useState('all');


      const [searchQuery, setSearchQuery] = useState('');

      const [selectedStatus, setSelectedStatus] = useState('all');

      const [selectedPriority, setSelectedPriority] = useState('all');

      const [selectedAssignee, setSelectedAssignee] = useState('all');


      const [isModalOpen, setIsModalOpen] = useState(false);

      const [editingTask, setEditingTask] = useState(null);

      const [submittingTask, setSubmittingTask] = useState(null);

      const [reviewingTask, setReviewingTask] = useState(null);

      const [taskToDelete, setTaskToDelete] = useState(null);


      // Refresh Lucide icon rendering on component updates

      useEffect(() => {

        if (window.lucide) {

          window.lucide.createIcons();

        }

      });


      // Authenticate User

      useEffect(() => {

        const { auth, signInAnonymously, signInWithCustomToken, onAuthStateChanged } = window.FirebaseServices;

        

        const initAuth = async () => {

          try {

            if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {

              await signInWithCustomToken(auth, __initial_auth_token);

            } else {

              await signInAnonymously(auth);

            }

          } catch (e) {

            console.error("Auth initialization error:", e);

          }

        };


        initAuth();


        const unsubscribe = onAuthStateChanged(auth, (currentUser) => {

          setUser(currentUser);

        });


        return () => unsubscribe();

      }, []);


      // Real-time Firestore Sync

      useEffect(() => {

        if (!user) return;


        const { db, appId, collection, onSnapshot } = window.FirebaseServices;

        // Path following Rule 1: /artifacts/{appId}/public/data/tasks

        const tasksCollectionRef = collection(db, 'artifacts', appId, 'public', 'data', 'tasks');


        setLoading(true);

        const unsubscribe = onSnapshot(tasksCollectionRef, (snapshot) => {

          const loadedTasks = snapshot.docs.map(doc => ({

            id: doc.id,

            ...doc.data()

          })).filter(task => !task.boardId || task.boardId === boardId);


          setTasks(loadedTasks);

          setLoading(false);

        }, (error) => {

          console.error("Real-time listener error:", error);

          setLoading(false);

        });


        return () => unsubscribe();

      }, [user, boardId]);


      // Save / Add Task to Firestore

      const handleSaveTask = async (taskData) => {

        if (!user) return;

        const { db, appId, collection, doc, setDoc, addDoc } = window.FirebaseServices;


        try {

          if (editingTask) {

            const taskDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'tasks', editingTask.id);

            await setDoc(taskDocRef, { ...editingTask, ...taskData, boardId }, { merge: true });

          } else {

            const tasksCollectionRef = collection(db, 'artifacts', appId, 'public', 'data', 'tasks');

            await addDoc(tasksCollectionRef, {

              ...taskData,

              boardId,

              evidenceFiles: [],

              createdAt: new Date().toISOString().split('T')[0]

            });

          }

        } catch (e) {

          console.error("Error saving task:", e);

        }


        setIsModalOpen(false);

        setEditingTask(null);

      };


      // Delete Task from Firestore

      const confirmDeleteTask = async () => {

        if (!user || !taskToDelete) return;

        const { db, appId, doc, deleteDoc } = window.FirebaseServices;


        try {

          const taskDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'tasks', taskToDelete.id);

          await deleteDoc(taskDocRef);

        } catch (e) {

          console.error("Error deleting task:", e);

        }


        setTaskToDelete(null);

      };


      // Handle Stage Move

      const handleMoveStage = async (taskId, newStatus) => {

        if (!user) return;

        const { db, appId, doc, updateDoc } = window.FirebaseServices;


        try {

          const taskDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'tasks', taskId);

          await updateDoc(taskDocRef, { status: newStatus });

        } catch (e) {

          console.error("Error moving stage:", e);

        }

      };


      // Handle Proof Submission with Cloud Upload

      const handleSubmitEvidence = async (taskId, submissionData) => {

        if (!user) return;

        const { db, storage, appId, doc, updateDoc, ref, uploadBytes, getDownloadURL } = window.FirebaseServices;


        try {

          const uploadedFileUrls = [];


          if (submissionData.newFiles && submissionData.newFiles.length > 0) {

            for (const fileObj of submissionData.newFiles) {

              if (fileObj.rawFile) {

                const storageRef = ref(storage, `evidence/${appId}/${Date.now()}_${fileObj.rawFile.name}`);

                await uploadBytes(storageRef, fileObj.rawFile);

                const downloadUrl = await getDownloadURL(storageRef);

                uploadedFileUrls.push({

                  id: `file-${Date.now()}`,

                  name: fileObj.rawFile.name,

                  type: fileObj.rawFile.type,

                  dataUrl: downloadUrl

                });

              } else if (fileObj.dataUrl) {

                uploadedFileUrls.push(fileObj);

              }

            }

          }


          const targetTask = tasks.find(t => t.id === taskId);

          const existingFiles = targetTask?.evidenceFiles || [];


          const taskDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'tasks', taskId);

          await updateDoc(taskDocRef, {

            status: submissionData.markCompletedImmediately ? 'completed' : 'in_review',

            evidenceUrl: submissionData.evidenceUrl || targetTask?.evidenceUrl || '',

            evidenceFiles: [...existingFiles, ...uploadedFileUrls],

            submitterNotes: submissionData.submitterNotes,

            submittedAt: new Date().toLocaleString(),

            submittedBy: submissionData.submittedBy || 'Team Member'

          });

        } catch (e) {

          console.error("Error submitting evidence:", e);

        }


        setSubmittingTask(null);

      };


      const handleApproveReject = async (taskId, newStatus) => {

        if (!user) return;

        const { db, appId, doc, updateDoc } = window.FirebaseServices;


        try {

          const taskDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'tasks', taskId);

          await updateDoc(taskDocRef, { status: newStatus });

        } catch (e) {

          console.error("Error updating status:", e);

        }


        setReviewingTask(null);

      };


      const filteredTasks = useMemo(() => {

        return tasks.filter(task => {

          if (activeTab === 'recurring' && !task.isRecurring) return false;

          if (activeTab === 'recurring' && recurringFrequency !== 'all' && task.frequency !== recurringFrequency) {

            return false;

          }


          const matchesSearch = 

            (task.title || '').toLowerCase().includes(searchQuery.toLowerCase()) ||

            (task.description || '').toLowerCase().includes(searchQuery.toLowerCase()) ||

            (task.assignee || '').toLowerCase().includes(searchQuery.toLowerCase());

          

          const matchesStatus = selectedStatus === 'all' || task.status === selectedStatus;

          const matchesPriority = selectedPriority === 'all' || task.priority === selectedPriority;

          const matchesAssignee = selectedAssignee === 'all' || task.assignee === selectedAssignee;


          return matchesSearch && matchesStatus && matchesPriority && matchesAssignee;

        });

      }, [tasks, activeTab, recurringFrequency, searchQuery, selectedStatus, selectedPriority, selectedAssignee]);


      return (

        <div className="min-h-screen flex flex-col font-sans pb-12">

          {/* Header Bar */}

          <header className="bg-white border-b border-slate-200 sticky top-0 z-30 shadow-xs">

            <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 h-16 flex items-center justify-between">

              <div className="flex items-center space-x-3">

                <div className="bg-indigo-600 text-white p-2 rounded-xl shadow-md">

                  <i data-lucide="layers" className="w-5 h-5"></i>

                </div>

                <div>

                  <h1 className="text-lg sm:text-xl font-bold text-slate-900 tracking-tight flex items-center space-x-2">

                    <span>Collaborative Workflows</span>

                    <span className="bg-emerald-100 text-emerald-800 text-[10px] font-extrabold px-2 py-0.5 rounded-full uppercase border border-emerald-200">

                      Live Sync

                    </span>

                  </h1>

                  <p className="text-xs text-slate-500 font-medium">Shared Real-Time Task & Deliverable Tracker</p>

                </div>

              </div>


              {/* Room / Board Switcher */}

              <div className="flex items-center space-x-2">

                <div className="hidden md:flex items-center space-x-1.5 bg-slate-100 p-1 rounded-xl border border-slate-200">

                  <span className="text-xs font-bold text-slate-500 pl-2">Board Code:</span>

                  <input

                    type="text"

                    value={boardInput || boardId}

                    onChange={(e) => setBoardInput(e.target.value)}

                    placeholder="Enter Board ID"

                    className="text-xs font-bold bg-white px-2 py-1 rounded-lg border border-slate-200 text-slate-800 focus:outline-none w-32"

                  />

                  <button

                    onClick={() => boardInput && setBoardId(boardInput.trim())}

                    className="bg-indigo-600 hover:bg-indigo-700 text-white text-xs font-bold px-2.5 py-1 rounded-lg transition-colors"

                  >

                    Join

                  </button>

                </div>


                <button

                  onClick={() => {

                    setEditingTask(null);

                    setIsModalOpen(true);

                  }}

                  className="bg-indigo-600 hover:bg-indigo-700 text-white px-3.5 py-2 rounded-xl text-xs font-bold flex items-center space-x-1.5 shadow-xs cursor-pointer transition-all"

                >

                  <i data-lucide="plus" className="w-4 h-4"></i>

                  <span>Create Shared Task</span>

                </button>

              </div>

            </div>

          </header>


          {/* Main Area */}

          <main className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-6 flex-1 w-full flex flex-col space-y-6">

            

            {/* View & Tab Switcher */}

            <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3 border-b border-slate-200 pb-3">

              <div className="flex space-x-4">

                <button

                  onClick={() => setActiveTab('main')}

                  className={`text-sm font-bold pb-1 border-b-2 transition-all cursor-pointer ${

                    activeTab === 'main' ? 'border-indigo-600 text-indigo-600' : 'border-transparent text-slate-500'

                  }`}

                >

                  All Shared Tasks ({tasks.length})

                </button>

                <button

                  onClick={() => setActiveTab('recurring')}

                  className={`text-sm font-bold pb-1 border-b-2 transition-all cursor-pointer ${

                    activeTab === 'recurring' ? 'border-indigo-600 text-indigo-600' : 'border-transparent text-slate-500'

                  }`}

                >

                  Recurring (Daily / Weekly / Monthly)

                </button>

              </div>


              {activeTab === 'main' && (

                <div className="flex items-center space-x-2 bg-slate-200/60 p-1 rounded-lg self-start sm:self-auto">

                  <button

                    onClick={() => setViewMode('kanban')}

                    className={`px-3 py-1 rounded-md text-xs font-bold ${

                      viewMode === 'kanban' ? 'bg-white text-indigo-600 shadow-xs' : 'text-slate-600'

                    }`}

                  >

                    Kanban Board

                  </button>

                  <button

                    onClick={() => setViewMode('list')}

                    className={`px-3 py-1 rounded-md text-xs font-bold ${

                      viewMode === 'list' ? 'bg-white text-indigo-600 shadow-xs' : 'text-slate-600'

                    }`}

                  >

                    Table List

                  </button>

                </div>

              )}

            </div>


            {/* Filter Controls */}

            <div className="bg-white p-3.5 rounded-xl border border-slate-200 shadow-2xs flex flex-wrap items-center justify-between gap-3">

              <input

                type="text"

                placeholder="Search tasks, proof links, names..."

                value={searchQuery}

                onChange={(e) => setSearchQuery(e.target.value)}

                className="text-xs px-3 py-1.5 border border-slate-200 rounded-lg focus:outline-none w-full sm:w-64"

              />


              <div className="flex flex-wrap items-center gap-2">

                <select

                  value={selectedStatus}

                  onChange={(e) => setSelectedStatus(e.target.value)}

                  className="text-xs border border-slate-200 bg-white rounded-lg px-2.5 py-1.5 text-slate-700 font-medium"

                >

                  <option value="all">All Stages</option>

                  {STAGES.map(s => <option key={s.id} value={s.id}>{s.name}</option>)}

                </select>


                <select

                  value={selectedPriority}

                  onChange={(e) => setSelectedPriority(e.target.value)}

                  className="text-xs border border-slate-200 bg-white rounded-lg px-2.5 py-1.5 text-slate-700 font-medium"

                >

                  <option value="all">All Priorities</option>

                  {Object.keys(PRIORITIES).map(p => <option key={p} value={p}>{PRIORITIES[p].label}</option>)}

                </select>

              </div>

            </div>


            {/* Main Task View Rendering */}

            {loading ? (

              <div className="text-center py-16 text-slate-500 font-medium">

                Syncing shared board with cloud...

              </div>

            ) : filteredTasks.length === 0 ? (

              <div className="bg-white rounded-2xl p-12 text-center border border-slate-200">

                <p className="text-slate-600 font-bold text-sm">No tasks found on this board.</p>

                <p className="text-slate-400 text-xs mt-1">Click "Create Shared Task" to add work for your team.</p>

              </div>

            ) : viewMode === 'kanban' ? (

              <KanbanView

                tasks={filteredTasks}

                stages={STAGES}

                onMoveStage={handleMoveStage}

                onOpenSubmitModal={setSubmittingTask}

                onOpenReviewModal={setReviewingTask}

                onEditTask={(task) => { setEditingTask(task); setIsModalOpen(true); }}

                onDeleteTask={setTaskToDelete}

              />

            ) : (

              <TableView

                tasks={filteredTasks}

                stages={STAGES}

                onMoveStage={handleMoveStage}

                onOpenSubmitModal={setSubmittingTask}

                onOpenReviewModal={setReviewingTask}

                onEditTask={(task) => { setEditingTask(task); setIsModalOpen(true); }}

                onDeleteTask={setTaskToDelete}

              />

            )}

          </main>


          {/* Modal Components */}

          {isModalOpen && (

            <TaskModal

              task={editingTask}

              stages={STAGES}

              onClose={() => { setIsModalOpen(false); setEditingTask(null); }}

              onSave={handleSaveTask}

            />

          )}


          {submittingTask && (

            <SubmitProofModal

              task={submittingTask}

              onClose={() => setSubmittingTask(null)}
